1. VISÃO GERAL DO SISTEMA DE AUTENTICAÇÃO
================================================================================

O OptmaMenu possui DOIS sistemas de autenticação distintos:

A) AUTENTICAÇÃO DE ADMINISTRADORES/LOJISTAS:
   - Usa Supabase Auth (auth.users)
   - Baseada em e-mail e senha
   - Gerenciada via useAuthStore
   - Protege rotas /admin/*

B) AUTENTICAÇÃO DE CLIENTES/CONSUMIDORES:
   - Sistema próprio na tabela customers
   - Baseada em telefone + OTP (código SMS/WhatsApp)
   - Opcionalmente com senha (bcrypt)
   - Gerenciada via useCustomerAuth + AuthService
   - Para acesso ao cardápio online

import { create } from 'zustand';

interface AuthState {

    session: Session | null;

    setSession: (session: Session | null) => void;
    setUser: (user: User | null) => void;
    setProfile: (profile: Profile | null) => void;

export const useAuthStore = create<AuthState>((set) => ({

    session: null,
    profile: null,

    setLoading: (loading) => set({ loading }),
}));

import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { Customer } from '@/types';


// Re-export Customer type for backward compatibility
export type { Customer };

interface CustomerAuthState {
    customer: Customer | null;
    token: string | null; // For future real Auth integration
    isAuthenticated: boolean;

    // Actions
    login: (customer: Customer) => void;
    logout: () => void;
    updatePoints: (newPoints: number) => void;
}

export const useCustomerAuth = create<CustomerAuthState>()(
    persist(
        (set) => ({
            customer: null,
            token: null,
            isAuthenticated: false,

            login: (customer) => set({ customer, isAuthenticated: true }),

            logout: () => set({ customer: null, isAuthenticated: false }),

            updatePoints: (newPoints) => set((state) => ({
                customer: state.customer ? { ...state.customer, loyalty_points: newPoints } : null
            }))
        }),
        {
            name: 'optma-customer-auth', // LocalStorage key
            partialize: (state) => ({ customer: state.customer, isAuthenticated: state.isAuthenticated }), // Persist only necessary data
        }
    )
);

import { supabase } from '@/lib/supabase';
import { useCustomerAuth } from '@/store/useCustomerAuth';
import bcrypt from 'bcryptjs';
import { NotificationService } from '@/services/notificationService';

export const AuthService = {

    // 0. Check User Status
    async checkStatus(phone: string, storeId: string) {

        const digits = phone.replace(/\D/g, '');
        const { data, error } = await supabase
            .from('customers')
            .select('*')
            .eq('phone', digits)
            .eq('store_id', storeId)
            .maybeSingle();

        if (error) {

            throw error;
        }


        return {
            exists: !!data,
            hasPassword: !!data?.password_hash,
            customer: data
        };
    },

    async sendOtp(phone: string, storeId: string) {

        const digits = phone.replace(/\D/g, '');
        const { data, error } = await supabase.rpc('send_customer_otp', {
            p_phone: digits,
            p_store_id: storeId
        });

        if (error) {

            throw error;
        }

        return data;
    },

    async verifyOtp(phone: string, otp: string, storeId: string) {

        const digits = phone.replace(/\D/g, '');
        const { data, error } = await supabase.rpc('verify_customer_otp', {
            p_phone: digits,
            p_otp: otp,
            p_store_id: storeId
        });

        if (error) {

            throw error;
        }


        if (data.isValid) {
            if (data.customer) {
                useCustomerAuth.getState().login(data.customer);
            }
            return {
                valid: true,
                isNewUser: data.isNewUser,
                customer: data.customer
            };
        }

        throw new Error('Código inválido ou expirado.');
    },

    // 3. Complete Registration (for new users - detailed profile comes later)
    async registerUser(data: { phone: string, storeId: string, storeName: string, nickname: string, password?: string, marketingConsent: boolean, email?: string, contactPreference?: 'whatsapp' | 'sms', birthDate?: string, loyaltyOptIn?: boolean }) {

        try {
            let passwordHash = null;
            if (data.password) {
                passwordHash = await bcrypt.hash(data.password, 10);
            }

            const insertData = {
                phone: data.phone,
                store_id: data.storeId,
                nickname: data.nickname,
                email: data.email || null,
                birth_date: data.birthDate || null,
                password_hash: passwordHash,
                marketing_consent: data.marketingConsent,
                contact_preference: data.contactPreference || 'whatsapp',
                loyalty_opt_in: data.loyaltyOptIn ?? false, // Default to false per new multi-step flow
                status: 'active'
            };

            const { data: newCustomer, error } = await supabase
                .from('customers')
                .insert(insertData)
                .select()
                .maybeSingle();

            if (error) {
                throw error;
            }

            // WELCOME MESSAGE
            try {
                await NotificationService.sendWelcome(newCustomer.id, data.storeId, data.storeName);
            } catch (notifError) {
                console.error('Failed to send welcome notification:', notifError);
                // Continue execution, do not fail registration
            }

            useCustomerAuth.getState().login(newCustomer);
            return newCustomer;
        } catch (error: any) {
            throw new Error(error.message || 'Erro ao criar conta.');
        }
    },

    // 4. Update Profile (Full Registration)
    async updateProfile(customerId: string, data: { fullName?: string, cpf?: string, email?: string, birthDate?: string }) {
        const { data: updated, error } = await supabase
            .from('customers')
            .update({
                full_name: data.fullName,
                cpf: data.cpf,
                email: data.email,
                birth_date: data.birthDate
            })
            .eq('id', customerId)
            .select()
            .maybeSingle();

        if (error) throw new Error('Erro ao atualizar perfil.');

        // Update local session
        useCustomerAuth.getState().login(updated);
        return updated;
    },

    // 5. Login with Password
    async loginWithPassword(phone: string, password: string, storeId: string) {
        const cleanPhone = phone.replace(/\D/g, '');

        const { data: customer, error } = await supabase
            .from('customers')
            .select('*')
            .eq('phone', cleanPhone)
            .eq('store_id', storeId)
            .maybeSingle();

        if (error || !customer) {
            throw new Error('Usuário não encontrado ou senha incorreta.');
        }

        if (!customer.password_hash) {
            throw new Error('Você ainda não tem senha. Entre com código OTP e crie uma.');
        }

        const isValid = await bcrypt.compare(password, customer.password_hash);
        if (!isValid) {
            throw new Error('Senha incorreta.');
        }

        useCustomerAuth.getState().login(customer);
        return customer;
    },

    // 6. Reset Password (called after OTP verification)
    async resetPassword(phone: string, storeId: string, newPassword: string) {
        const cleanPhone = phone.replace(/\D/g, '');
        const hash = await bcrypt.hash(newPassword, 10);

        const { error } = await supabase
            .from('customers')
            .update({ password_hash: hash })
            .eq('phone', cleanPhone)
            .eq('store_id', storeId);

        if (error) throw new Error('Erro ao redefinir senha.');
        return true;
    }
};

import { useAuthStore } from '@/store/useAuthStore';
import { useEffect, useState } from 'react';

export function ProtectedRoute() {
    const { session, loading } = useAuthStore();
    const location = useLocation();
    const [isChecking, setIsChecking] = useState(true);

    useEffect(() => {
        // Pequeno delay para evitar flicker
        const timer = setTimeout(() => {
            setIsChecking(false);
        }, 300);

        return () => clearTimeout(timer);
    }, []);

    if (loading || isChecking) {
        return (
            <div className="min-h-screen flex items-center justify-center bg-gray-50 dark:bg-gray-900">
                <div className="flex flex-col items-center">
                    <div className="relative">
                        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-[#21A896]"></div>
                        <div className="absolute top-0 left-0 w-12 h-12 rounded-full border-2 border-[#21A896] opacity-20"></div>
                    </div>
                    <p className="mt-4 text-gray-600 dark:text-gray-300 font-candara">
                        Verificando sua autenticação...
                    </p>
                </div>
            </div>
        );
    }

    if (!session) {
        return <Navigate to="/login" state={{ from: location }} replace />;
    }

    return <Outlet />;
}

import { useEffect } from 'react';
import { BrowserRouter } from 'react-router-dom';
import { useAuthStore } from '@/store/useAuthStore';
import { supabase } from '@/lib/supabase';
import AppRoutes from '@/AppRoutes';
import CookieConsent from '@/components/common/CookieConsent';
import { Toaster } from 'sonner';

function App() {
  const { setSession, setLoading, setProfile } = useAuthStore();

  useEffect(() => {
    // Check active session
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      if (session) {
        // Fetch profile
        supabase
          .from('profiles')
          .select('*')
          .eq('id', session.user.id)
          .maybeSingle()
          .then(({ data }) => {
            if (data) setProfile(data);
          });
      }
      setLoading(false);
    });

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
      if (session) {
        supabase
          .from('profiles')
          .select('*')
          .eq('id', session.user.id)
          .maybeSingle()
          .then(({ data }) => {
            if (data) setProfile(data);
          });
      }
    });

    return () => subscription.unsubscribe();
  }, [setSession, setLoading, setProfile]);

  return (
    <BrowserRouter>
      <AppRoutes />
      <CookieConsent />
      <Toaster position="top-right" richColors />
    </BrowserRouter>
  );
}

export default App;

        </div>
    );
}

-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- PROFILES (Users)
create table profiles (
  id uuid references auth.users on delete cascade primary key,
  phone text unique,
  name text,
  address text,
  district text, -- Bairro
  city text default 'São João Nepomuceno',
  birthData date,
  cpf text,
  points integer default 0,
  accepted_policy boolean default false,
  accepted_promo boolean default false,
  is_admin boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Handle new user signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, phone, name)
  values (new.id, new.phone, new.raw_user_meta_data->>'full_name');
  return new;
end;
$$ language plpgsql security definer;

10. FLUXO DE AUTENTICAÇÃO - RESUMO
================================================================================

A) FLUXO DO ADMINISTRADOR/LOJISTA:
   1. Usuário acessa /signup
   2. Preenche dados pessoais + dados da loja
   3. supabase.auth.signUp() cria usuário em auth.users
   4. Trigger handle_new_user() cria registro em profiles
   5. Usuário confirma e-mail
   6. Redirecionado para /onboarding/create-store
   7. Cria loja na tabela stores
   8. Acessa /admin (protegido por ProtectedRoute)

B) FLUXO DO CLIENTE/CONSUMIDOR:
   1. Cliente acessa cardápio (/s/{slug})
   2. Faz checkout como convidado ou login
   3. Login via OTP (send_customer_otp + verify_customer_otp)
   4. Ou login com senha (bcrypt)
   5. AuthService.loginWithPassword() ou verifyOtp()
   6. Estado persistido em useCustomerAuth (localStorage)
